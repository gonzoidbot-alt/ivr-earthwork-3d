import * as THREE from 'three';
import {OrbitControls} from 'three/addons/controls/OrbitControls.js';
/*__AUDITED_DATA__*/

const PH = DATA.phases;
const VSCALE = 2.2, ZBASE = 2220;
const ST = DATA.stations;
const N = ST.length;
const spine = DATA.spine;
const ring = DATA.pavement_ring;
const egRings = DATA.eg_rings;

let minX=1e9,maxX=-1e9,minZ=1e9,maxZ=-1e9;
for(const c of ST){minX=Math.min(minX,c.x);maxX=Math.max(maxX,c.x);minZ=Math.min(minZ,c.y);maxZ=Math.max(maxZ,c.y);}
for(const p of ring){minX=Math.min(minX,p[0]);maxX=Math.max(maxX,p[0]);minZ)Math.min(minZ,p[1]);maxZ=Math.max(maxZ,p[1]);}

/* Coordinate transforms remain in the original model frame. */
const cx=(minX+maxX)/2, cz=(minZ+maxZ)/2;
const span=Math.max(maxX-minX,maxZ-minZ);
const WX=v=>v-cx, WZ=v=>v-cz;
const MZ=y=>-(y-cz);

/*__G_BINDING__*/
/*__GRID_HELPER__*/

const clamp=THREE.MathUtils.clamp;
const mix=THREE.MathUtils.lerp;
const reducedMotion=matchMedia('(prefers-reduced-motion:reduce)').matches;

let minElevation=Infinity,maxElevation=-Infinity;
for(let i=0;i<G.eg.length;i++){
  minElevation=Math.min(minElevation,G.eg[i],G.design[i]);
  maxElevation=Math.max(maxElevation,G.eg[i],G.design[i]);
}
for(const p of DATA.wall_daylight) maxElevation=Math.max(maxElevation,p[2]);
for(const p of DATA.tank_pad) maxElevation=Math.max(maxElevation,p[2]);

const siteBounds=new THREE.Box3(
  new THREE.Vector3(WX(G.x0),(minElevation-ZBASE)*VSCALE,MZ(G.z1)),
  new THREE.Vector3(WX(G.x1),(maxElevation-ZBASE)*VSCALE+1,MZ(G.z0))
);
const siteSphere=siteBounds.getBoundingSphere(new THREE.Sphere());
const siteCenter=siteSphere.center;
const siteR=siteSphere.radius;
const groundY=siteBounds.min.y-14;

const scene=new THREE.Scene();
scene.background=new THREE.Color(0x0b0d12);
scene.fog=new THREE.Fog(0x0b0d12,siteR*4,siteR*11);

const camera=new THREE.PerspectiveCamera(48,innerWidth/innerHeight,1,siteR*50);
const renderer=new THREE.WebGLRenderer({antialias:true,powerPreference:'high-performance'});
renderer.setPixelRatio(Math.min(devicePixelRatio,2));
renderer.setSize(innerWidth,innerHeight);
renderer.shadowMap.enabled=true;
renderer.shadowMap.type=THREE.PCFSoftShadowMap;
renderer.outputColorSpace=THREE.SRGBColorSpace;
renderer.toneMapping=THREE.ACESFilmicToneMapping;
renderer.toneMappingExposure=1.1;
document.getElementById('app').appendChild(renderer.domElement);
renderer.domElement.tabIndex=0;
renderer.domElement.setAttribute('aria-label','Diamond Heights three-dimensional terrain model');

const controls=new OrbitControls(camera,renderer.domElement);
controls.enableDamping=true;
controls.dampingFactor=.06;
controls.maxPolarAngle=Math.PI*.465;
controls.minDistance=80;
controls.maxDistance=siteR*12;
controls.zoomSpeed=.8;
controls.panSpeed=.8;
controls.autoRotate=false;
controls.autoRotateSpeed=.12;

function fitCam(){
  const R=siteSphere.radius;
  const center=siteSphere.center;
  const vfov=camera.fov*Math.PI/180;
  const hfov=2*Math.atan(Math.tan(vfov/2)*camera.aspect);
  const dist=R/Math.sin(Math.min(vfov,hfov)/2)*1.18;
  const el=32*Math.PI/180,az=-47*Math.PI/180;
  const dir=new THREE.Vector3(Math.sin(az)*Math.cos(el),Math.sin(el),Math.cos(az)*Math.cos(el));
  controls.autoRotate=false;
  controls.maxDistance=Math.max(siteR*12,dist*2.25);
  camera.far=Math.max(siteR*50,dist*10);
  camera.position.copy(center).addScaledVector(dir,dist);
  controls.target.copy(center);
  camera.lookAt(center);
  camera.updateProjectionMatrix();
  controls.update();

/* Deterministic procedural appearance. None of these functions change DATA. */
function hash2(ix,iz){
  let h=Math.imul(ix,374761393)+Math.imul(iz,668265263);
  h=Math.imul(h^(h>>>13),1274126177);
  return ((h^(h>>>16))>>>0)/4294967295;
}
function noise2(x,z){
  const ix=Math.floor(x),iz=Math.floor(z);
  let tx=x-ix,tz=z-iz;
  tx=tx*tx*(3-2*tx);tz=tz*tz*(3-2*tz);
  return mix(mix(hash2(ix,iz),hash2(ix+1,iz),tx),mix(hash2(ix,iz+1),hash2(ix+1,iz+1),tx),tz);
}
function fbm(x,z){return noise2(x,z)*.57+noise2(x*2.03+13,z*2.03-17)*.28+noise2(x*4.1-53,z*4.1+29)*.15;}
function surfaceTexture(seed,base,brightness){
  const size=128,data=new Uint8Array(size*size*4);
  for(let y=0;y<size;y++)for(let x=0;x<size;x++){
    const n=hash2(x+seed,y-seed);
    const grain=(n-.5)*brightness;
    const value=clamp(base+grain+(n>.978?-brightness*.8:0),0,255);
    const k=(y*size+x)*4;
    data[k]=data[k+1]=data[k+2]=value;data[k+3]=255;
  }
  const t=new THREE.DataTexture(data,size,size,THREE.RGBAFormat);
  t.wrapS=t.wrapT=THREE.RepeatWrapping;
  t.magFilter=THREE.LinearFilter;
  t.minFilter=THREE.LinearMipmapLinearFilter;
  t.generateMipmaps=true;
  t.anisotropy=Math.min(8,renderer.capabilities.getMaxAnisotropy());
  t.needsUpdate=true;
  return t;
}

/* Geometry utilities preserve every original outline coordinate in the source data. */
function cleanPlanRing(pts){
  const out=[];
  for(const p of pts){
    const last=out[out.length-1];
    if(last&&Math.hypot(last[0]-p[0],last[1]-p[1])<1e-9)continue;
    out.push([p[0],p[1],p[2]]);
  }
  if(out.length>2&&Math.hypot(out[0][0]-out.at(-1)[0],out[0][1]-out.at(-1)[1])<1e-9)out.pop();
  return out;
}

function inRing(x,z){

  for(let i=0,j=polygonRing.length-1;i<polygonRing.length;j=i++){

}
function area2(pts){
  let a=0;
  for(let i=0;i<pts.length;i++){

}

/* A small median-split segment index for procedural crown appearance only. */
function segmentTree(pts,closed){
  const segments=[];
  for(let i=0;i<pts.length-(closed?0:1);i++){
    const a=pts[i],b=pts[(i+1)%pts.length];
    if(Math.hypot(b[0]-a[0],b[1]-a[1])<1e-6)continue;
    segments.push({ax:a[0],z:a[1],dx:b[0]-a[0],dz:b[1]-a[1],minX:Math.min(a[0],b[0]),maxX:Math.max(a[0],b[0]),minZ:Math.min(a[1],b[1]),maxZ:Math.max(a[1],b[1])});
  }
  function build(items){
    const n={minX:Infinity,minZ:Infinity,maxX:-Infinity,maxZ:-Infinity};
    for(const s of items){n.minX=Math.min(n.minX,s.minX);n.maxX=Math.max(n.maxX,s.maxX);n.minZ=Math.min(n.minZ,s.minZ);n.maxZ=Math.max(n.maxZ,s.maxZ);}
    if(items.length<=8){n.items=items;return n;}
    const horizontal=n.maxX-n.minX>n.maxZ-n.minZ;
    items.sort((a,b)=>horizontal?(a.minX+a.maxX)-(b.minX+b.maxX):(a.minZ+a.maxZ)-(b.minZ+b.maxZ));
    const mid=items.length>>1;
    n.left=build(items.slice(0,mid));n.right=build(items.slice(mid));
    return n;
  }
  const root=build(segments);
  return function distance(x,z){
    let best=Infinity;
    function boxDist(n){
      const dx=Math.max(n.minX-x,0,x-n.maxX),dz=Math.max(n.minZ-z,0,z-n.maxZ);
      return dx*dx+dz*dz;
    }
    function visit(n){
      if(boxDist(n)>=best)return;
      if(n.items){
        for(const s of n.items){
          const t=clamp(((x-s.ax)*s.dx+(z-s.z)*s.dz)/(s.dx*s.dx+s.dz*s.dz),0,1);
          const dx=x-s.ax-t*s.dx,dz=z-s.z-t*s.dz;
          best=Math.min(best,dx*dx+dz*dz);
        }
      }else{
        const left first=boxDist(n.left)<=boxDist(n.right);
        if(left first){visit(n.left);visit(n.right);}
        else{visit(n.right);visit(n.left);}
      }
    }
    visit(root);
    return Math.sqrt(best);
  };
}

}

/* Conforming subdivision: every shared edge is split consistently. */
function sampledPolygon(pts,maxEdge,heightFunction){
  const verts=cleanPlanRing(pts);
  const shapePoints=verts.map(p=>new THREE.Vector2(p[0],p[1]));
  let faces=THREE.ShapeUtils.triangulateShape(shapePoints,[]);
  for(let pass=0;pass<19;pass++){
    const mids=new Map();
    const limit=maxEdge*maxEdge;
    function mark(a,b){
      const key=a<b?`${a}:${b}`:`${b}:${a}`;
      if(mids.has(key))return;
      const p=verts[a],q=verts[b],dx=p[0]-q[0],dz=p[1]-q[1];
      if(dx*dx+dz*dz>limit){
        mids.set(key,verts.length);
        verts.push([(p[0]+q[0])*.5,(p[1]+q[1])*.5,0]);
      }
    }
    for(const [a,b,c] of faces){mark(a,b);mark(b,c);mark(c,a);}
    if(!mids.size)break;
    const next=[];
    const mid=(a,b)=>mids.get(a<b?`${a}:${b}`:`${b}:${a}`);
    for(const [a,b,c] of faces){
      const ab=mid(a,b),bc=mid(b,c),ca=mid(c,a);
      const mask=(ab!==undefined?1:0)|(bc!==undefined?2:0)|(ca!==undefined?4:0);
      switch(mask){
        case 0:next.push([a,b,c]);break;
        case 1:next.push([a,ab,c],[ab,b,c]);break;
        case 2:next.push([a,b,bc],[a,bc,c]);break;
        case 3:next.push([ab,b,bc],[a,ab,bc],[a,bc,c]);break;
        case 4:next.push([a,b,ca],[ca,b,c]);break;
        case 5:next.push([a,ab,ca],[ab,b,c],[ab,c,ca]);break;
        case 6:next.push([bc,c,ca],[a,b,bc],[a,bc,ca]);break;
        case 7:next.push([a,ab,ca],[ab,b,bc],[ca,bc,c],[ab,bc,ca]);break;
      }
    }
    faces=next;
    if(faces.length>600000)throw Error('Surface tessellation exceeded the presentation mesh budget.');
  }
  const g=new THREE.BufferGeometry();
  const p=new Float32Array(verts.length*3),t=new Float32Array(verts.length*2);
  for(let i=0;i<verts.length;i++){
    const [x,z]=verts[i];
    p[i*3]=WX(x);p[i*3+1]=heightFunction(x,z);p[i*3+2]=MZ(z);
    t[i*2]=x/8;t[i*2+1]=z/8;
  }
  const indices=[];
  for(let f of faces){
    const a=verts[f[0]],b=verts[f[1]],c=verts[f[2]];
    if((b[0]-a[0])*(c[1]-a[1])-(b[1]-a[1])*(c[0]-a[0])<0)f=[f[0],f[2],f[1]];
    indices.push(...f);
  }
  g.setAttribute('position',new THREE.BufferAttribute(p,3));
  g.setAttribute('uv',new THREE.BufferAttribute(t,2));
  g.setIndex(indices);g.computeVertexNormals();g.computeBoundingSphere();
  return g;
}
function densify(pts,spacing,closed=false){
  const out=[];
  for(let i=0;i<pts.length-(closed?0:1);i++){
    const a=pts[i],b=pts[(i+1)%pts.length];
    const count=Math.max(1,Math.ceil(Math.hypot(b[0]-a[0],b[1]-a[1])/spacing));
    for(let k=0;k<count;k++){
      const t=k/count;
      out.push([mix(a[0],b[0],t),mix(a[1],b[1],t),mix(a[2]??0,b[2]??0,t)]);
    }
  }
  if(!closed&&pts.length)out.push([...pts.at(-1)]);
  return out;
}
function planNormal(pts,i,closed){
  const p=pts[i];
  const a=pts[closed?(i+pts.length-1)%pts.length:Math.max(0,i-1)];
  const b=pts[closed?(i+1)%pts.length:Math.min(pts.length-1,i+1)];
  let ax=p[0]-a[0],az=p[1]-a[1],bx=b[0]-p[0],bz=b[1]-p[1];
  let al=Math.hypot(ax,az),bl=Math.hypot(bx,bz);
  if(al<1e-8){ax=bx;az=bz;al=bl;}
  if(bl<1e-8){bx=ax;bz=az;bl=al;}
  ax/=al||1;az/=al||1;bx/=bl||1;bz/=bl||1;
  let nx=-az-bz,nz=ax+bx;
  const n=Math.hypot(nx,nz);
  if(n<.001){nx=-bz;nz=bx;}else{nx/=n;nz/=n;}
  const determinant=Math.max(.5,nx*(-bz)+nz*bx);
  return [nx/determinant,nz/determinant];
}
function flatRibbon(pts,width,heightFunc,{closed=false,offset=0,clipToPavement=false}={}){
  const p=[],uv=[],indices=[];
  let length=0;
  for(let i=0;i<pts.length;i++){
    const [nx,nz]=planNormal(pts,i,closed),q=pts[i];
    if(i)length+=Math.hypot(q[0]-pts[i-1][0],q[1]-pts[i-1][1]);
    for(const side of [-1,1]){
      const k=offset+side*width*.5,x=q[0]+nx*k,z=q[1]+nz*k;
      p.push(WX(x),heightFunc(x,z),MZ(z));
      uv.push(length/4,side>0?1:0);
    }
  }
  for(let i=0;i<pts.length-(closed?0:1);i++){
    const a=i*2,b=((i+1)%pts.length)*2;
    for(const tri of [[a,a+1,b],[a+1,b+1,b]]){
      if(clipToPavement){
        let x=0,z=0;
        for(const t of tri){x+=p[t*3]+cx;z+=-p[t*3+2]+cz;}
        if(!inRing(x/3,z/3))continue;
      }
      indices.push(...tri);
    }
  }
  const g=new THREE.BufferGeometry();
  g.setAttribute('position',new THREE.Float32BufferAttribute(p,3));
  g.setAttribute('uv',new THREE.Float32BufferAttribute(uv,2));
  g.setIndex(indices);g.computeVertexNormals();g.computeBoundingSphere();
  return g;
}
