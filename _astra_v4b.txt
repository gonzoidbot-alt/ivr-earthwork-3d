    const fx = clamp((px-G.x0)/(G.x1-G.x0),0,1)*(G.nx-1);
    const fz = clamp((pz-G.z0)/(G.z1-G.z0),0,1)*(G.nz-1);

    const iz = Math.min(G.nz-2,Math.floor(fz));

    const a=G.eg,b=G.design;
    return {
      eg:a[i00]*(1-tx)*(1-tz)+a[i10]*tx*(1-tz)+a[i01]*(1-tx)*tz+a[i11]*tx*tz,
      dz:b[i00]*(1-tx)*(1-tz)+b[i10]*tx*(1-tz)+b[i01]*(1-tx)*tz+b[i11]*tx*tz
    };
  }

  function hash(x,y){
    let h=Math.imul(x|0,374761393)^Math.imul(y|0,668265263);
    h=Math.imul(h^(h>>>13),1274126177);
    return ((h^(h>>>16))>>>0)/4294967295;
  }
  function noise(x,y){
    const ix=Math.floor(x),iy=Math.floor(y);
    const u=smooth(x-ix),v=smooth(y-iy);
    return lerp(lerp(hash(ix,iy),hash(ix+1,iy),u),lerp(hash(ix,iy+1),hash(ix+1,iy+1),u),v);
  }
  function fbm(x,y){
    return noise(x,y)*.58+noise(x*2.07+17,y*2.07-13)*.28+noise(x*4.19-9,y*4.19+21)*.14;
  }
  function makeGrain(){
    const size=128,bytes=new Uint8Array(size*size*4);

        const value=Math.round(193+n*62);
        bytes[k]=bytes[k+1]=bytes[k+2]=value;
        bytes[k+3]=255;
      }
    }
    const texture=new THREE.DataTexture(bytes,size,size,THREE.RGBAFormat);
    texture.wrapS=texture.wrapT=THREE.RepeatWrapping;
    texture.magFilter=THREE.LinearFilter;
    texture.minFilter=THREE.LinearMipmapLinearFilter;
    texture.generateMipmaps=true;
    texture.anisotropy=Math.min(8,renderer.capabilities.getMaxAnisotropy());
    texture.needsUpdate=true;
    return texture;
  }

  // ---------- Geometry helpers ----------
  function densify(points,step=6,closed=false){
    const clean=cleanPath(points,closed),out=[];
    if (clean.length<2) return clean.slice();
    const count=closed?clean.length:clean.length-1;
    for (let i=0;i<count;i++){
      const a=clean[i],b=clean[(i+1)%clean.length];
      const steps=Math.max(1,Math.ceil(Math.hypot(b[0]-a[0],b[1]-a[1])/step));
      for (let j=0;j<steps;j++){
        const t=j/steps;
        out.push([lerp(a[0],b[0],t),lerp(a[1],b[1],t),
          Number.isFinite(a[2])&&Number.isFinite(b[2])?lerp(a[2],b[2],t):undefined]);
      }
    }
    if (!closed) out.push(clean.at(-1));
    return out;
  }

  // Shared-edge adaptive subdivision keeps the entire concave pavement footprint,
  // including its bulb, watertight while resolving the elevation grid.
  function tessellatePolygon(points,spacing=6){
    const shape=new THREE.Shape(cleanPath(points,true).map(p=>new THREE.Vector2(p[0],p[1])));
    const original=new THREE.ShapeGeometry(shape);
    const vertices=[];
    const p=original.attributes.position;
    for (let i=0;i<p.count;i++) vertices.push([p.getX(i),p.getY(i)]);
    let indices=Array.from(original.index.array);
    original.dispose();
    const spacing2=spacing*spacing;
    for (let pass=0;pass<16;pass++){
      const mids=new Map();
      function midpoint(a,b){
        const va=vertices[a],vb=vertices[b];
        if ((va[0]-vb[0])**2+(va[1]-vb[1])**2<=spacing2) return -1;
        const key=a<b?a+':'+b:b+':'+a;
        if (mids.has(key)) return mids.get(key);
        const m=vertices.length;
        vertices.push([(va[0]+vb[0])/2,(va[1]+vb[1])/2]);
        mids.set(key,m);
        return m;
      }
      const next=[];
      for (let i=0;i<indices.length;i+=3){
        const a=indices[i],b=indices[i+1],c=indices[i+2];
        const ab=midpoint(a,b),bc=midpoint(b,c),ca=midpoint(c,a);
        const mask=(ab>=0?1:0)|(bc>=0?2:0)|(ca>=0?4:0);
        switch(mask){
          case 0:next.push(a,b,c);break;
          case 1:next.push(a,ab,c,ab,b,c);break;
          case 2:next.push(b,bc,a,bc,c,a);break;
          case 4:next.push(c,ca,b,ca,a,b);break;
          case 3:next.push(b,bc,ab,a,ab,c,ab,bc,c);break;
          case 6:next.push(c,ca,bc,b,bc,a,bc,ca,a);break;
          case 5:next.push(a,ab,ca,c,ca,b,ca,ab,b);break;
          case 7:next.push(a,ab,ca,ab,b,bc,ca,bc,c,ab,bc,ca);break;
        }
      }
      indices=next;
      if (!mids.size || vertices.length>180000) break;
    }
    return {vertices,indices};
  }

  // Plan-distance BVH only: this never interpolates elevations.
  function makeSegmentTree(points,closed=false){
    const segments=[];
    for (let i=0;i<(closed?points.length:points.length-1);i++){
      const a=points[i],b=points[(i+1)%points.length];
      segments.push({a,b,minX:Math.min(a[0],b[0]),maxX:Math.max(a[0],b[0]),minY:Math.min(a[1],b[1]),maxY:Math.max(a[1],b[1])});
    }
    function build(items){
      const node={minX:Infinity,minY:Infinity,maxX:-Infinity,maxY:-Infinity};
      for (const s of items){
        node.minX=Math.min(node.minX,s.minX);node.minY=Math.min(node.minY,s.minY);
        node.maxX=Math.max(node.maxX,s.maxX);node.maxY=Math.max(node.maxY,s.maxY);
      }
      if (items.length<=8){node.items=items;return node;}
      const axis=node.maxX-node.minX>node.maxY-node.minY?0:1;
      items.sort((a,b)=>(a.a[axis]+a.b[axis])-(b.a[axis]+b.b[axis]));
      const middle=items.length>>1;
      node.left=build(items.slice(0,middle));node.right=build(items.slice(middle));
      return node;
    }
    const root=build(segments);
    function boxDistance(node,x,y){
      const dx=Math.max(node.minX-x,0,x-node.maxX),dy=Math.max(node.minY-y,0,y-node.maxY);
      return dx*dx+dy*dy;
    }
    return (x,y)=>{
      let best=Infinity;
      function visit(node){
        if (boxDistance(node,x,y)>=best) return;
        if (node.items){
          for (const s of node.items){
            const dx=s.b[0]-s.a[0],dy=s.b[1]-s.a[1];
            const t=clamp(((x-s.a[0])*dx+(y-s.a[1])*dy)/(dx*dx+dy*dy||1),0,1);
            best=Math.min(best,(x-s.a[0]-t*dx)**2+(y-s.a[1]-t*dy)**2);
          }
        } else {
          const dl=boxDistance(node.left,x,y),dr=boxDistance(node.right,x,y);
          if (dl<dr){visit(node.left);visit(node.right);}else{visit(node.right);visit(node.left);}
        }
      }
      visit(root);
      return Math.sqrt(best);
    };
  }

  const centerDistance=makeSegmentTree(spine.length>1?spine:ring,false);
  const pavementDistance=makeSegmentTree(ring,true);
  function roadElevation(px,py){
    const g0=gridElev(px,py);
    const dCenter=centerDistance(px,py),dEdge=pavementDistance(px,py);
    const ratio=dCenter/Math.max(.001,dCenter+dEdge);
    const crown=.16*Math.max(0,1-ratio*ratio);
    return elevationY(g0.dz)+.65+crown*VSCALE;
  }

  function polygonSurface(points,spacing,heightFunction){
    const {vertices,indices}=tessellatePolygon(points,spacing);
    const positions=new Float32Array(vertices.length*3);
    const cols=new Float32Array(vertices.length*3);
    const uv=new Float32Array(vertices.length*2);
    for (let i=0;i<vertices.length;i++){
      const [x,y]=vertices[i],k=i*3;
      positions[k]=WX(x);positions[k+1]=heightFunction(x,y);positions[k+2]=MZ(y);
      const shade=.9+.10*noise(x*.23,y*.23);
      cols[k]=cols[k+1]=cols[k+2]=shade;
      uv[i*2]=x/9;uv[i*2+1]=y/9;
    }
    const geo=new THREE.BufferGeometry();
    geo.setAttribute('position',new THREE.BufferAttribute(positions,3));

  function contourTriangle(a,b,c){
    const low=Math.min(a[2],b[2],c[2]),high=Math.max(a[2],b[2],c[2]);
    for (let level=Math.ceil(low/contourInterval)*contourInterval;level<=high;level+=contourInterval){
      const crossings=[];
      for (const [p,q] of [[a,b],[b,c],[c,a]]){
        if ((p[2]<level&&q[2]>=level)||(q[2]<level&&p[2]>=level)){
          const t=(level-p[2])/(q[2]-p[2]);
          crossings.push([lerp(p[0],q[0],t),lerp(p[1],q[1],t)]);
        }
      }
      if (crossings.length!==2) continue;
      const p=crossings[0],q=crossings[1];
      if (Math.hypot(q[0]-p[0],q[1]-p[1])<.001) continue;
      const output=Math.abs(level/25-Math.round(level/25))<.001?majorContours:minorContours;
      const steps=Math.max(2,Math.ceil(Math.hypot(q[0]-p[0],q[1]-p[1])/3));
      for (let i=0;i<steps;i++){
        for (const t of [i/steps,(i+1)/steps]){
          const x=lerp(p[0],q[0],t),y=lerp(p[1],q[1],t);
          output.push(WX(x),elevationY(gridElev(x,y).dz)+.24,MZ(y));
        }
      }
    }

      });
    }

    currentProgress=clamp(lerp(previous,progressEnds[idx],t),0,1);

  function updateFlythrough(dt){
    if (!playing||!followInput.checked||!flyCurve){
      controls.enabled=true;
      return false;
    }
    controls.enabled=false;
    controls.autoRotate=false;
    const progress=clamp((phaseIdx+phaseT)/PH.length,0,1);
    const u=lerp(.015,.96,progress);
    flyCurve.getPointAt(u,flyPosition);
    flyCurve.getPointAt(Math.min(.995,u+.035),flyAhead);
    flyCurve.getTangentAt(u,flyTangent);
    const sample=gridElev(flyPosition.x+cx,cz-flyPosition.z);
    const visibleHeight=elevationY(lerp(sample.eg,sample.dz,clamp(lastMorph,0,1)));
    desiredCamera.copy(flyPosition).addScaledVector(flyTangent,-46);
    desiredCamera.y=Math.max(desiredCamera.y,visibleHeight)+85;
    const cameraGround=gridElev(desiredCamera.x+cx,cz-desiredCamera.z);
    desiredCamera.y=Math.max(desiredCamera.y,elevationY(lerp(cameraGround.eg,cameraGround.dz,lastMorph))+48);
    desiredTarget.copy(flyAhead);
    const aheadGround=gridElev(flyAhead.x+cx,cz-flyAhead.z);
    desiredTarget.y=Math.max(desiredTarget.y,elevationY(lerp(aheadGround.eg,aheadGround.dz,lastMorph)))+12;
    const ease=1-Math.exp(-dt*2.4);
    camera.position.lerp(desiredCamera,ease);
    controls.target.lerp(desiredTarget,ease);
    camera.lookAt(controls.target);
    return true;
  }

  let last=performance.now(),phaseAccumulator=0;
  const cameraDirection=new THREE.Vector3();
  function tick(now){
    requestAnimationFrame(tick);
    const dt=Math.min(.1,(now-last)/1000);
    last=now;
    if (document.hidden) return;

    if (playing){
      phaseT+=dt*.35;
      if (phaseT>=1){
        phaseT=0;phaseIdx++;
        if (phaseIdx>=PH.length){
          phaseIdx=PH.length-1;phaseT=1;playing=false;
          controls.enabled=true;
          $('play').textContent='▶ REPLAY SEQUENCE';
        }
        applyPhase(phaseIdx,phaseT);
        phaseAccumulator=0;
      } else {
        phaseAccumulator+=dt;
        // Dense terrain normals update at 30 Hz; camera and rendering stay fluid.
        if (phaseAccumulator>=1/30){
          applyPhase(phaseIdx,phaseT);
          phaseAccumulator=0;
        }
      }
    }
    const following=updateFlythrough(dt);
    if (!following) controls.update(dt);
    sky.position.copy(camera.position);
    camera.getWorldDirection(cameraDirection);
    const northAngle=Math.atan2(cameraDirection.x,-cameraDirection.z);
    $('compass-arrow').style.transform='rotate('+(-northAngle*180/Math.PI).toFixed(2)+'deg)';
    const status=following?'FOLLOW ROAD':controls.autoRotate&&!userInteracted?'SLOW ORBIT · INTERACT TO EXPLORE':'SITE OVERVIEW';
    if ($('camera-status').textContent!==status) $('camera-status').textContent=status;
    renderer.render(scene,camera);
  }

  applyPhase(0,1);
